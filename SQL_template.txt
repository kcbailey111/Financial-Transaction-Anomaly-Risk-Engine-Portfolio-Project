-- View all records in your table
SELECT * FROM your_table_name;

-- Filter for high-risk or flagged anomalies (common for a risk engine project)
SELECT 
    customer_id,
    amount,
    risk_score
FROM your_table_name
WHERE is_flagged = TRUE
ORDER BY amount DESC;

-- Aggregate data to get totals and averages by flag status
SELECT 
    is_flagged,
    COUNT(*) AS total_transactions,
    SUM(amount) AS total_amount,
    ROUND(AVG(risk_score)::numeric, 2) AS average_risk_score
FROM your_table_name
GROUP BY is_flagged;
